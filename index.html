Repli : mots-cles du nom, pour les ETF hors dictionnaire var hay = (nom + ' ' + code).toUpperCase(); for (var i = 0; i < PROFILES_ETF.length; i++) { for (var j = 0; j < PROFILES_ETF[i].m.length; j++) { if (hay.indexOf(PROFILES_ETF[i].m[j]) !== -1) return PROFILES_ETF[i]; } } return null; } // Exceptions : mnemoniques Euronext differents du ticker Yahoo var CODES_YAHOO = { 'AI': 'AI.PA', 'AL': 'AI.PA', 'FP': 'TTE.PA', 'WLB': 'WLN.PA', 'BI': 'BIM.PA' }; // Zone geographique deduite du suffixe du ticker Yahoo (.PA -> France...) function zoneDepuisSuffixe(c) { var s = String(c).toUpperCase(); if (s.indexOf('.') === -1) return 'USA'; var suf = s.substring(s.indexOf('.')); if (suf === '.PA' || suf === '.NX') return 'France'; if (suf === '.AS') return 'Pays-Bas'; if (suf === '.L') return 'Royaume-Uni'; if (suf === '.DE' || suf === '.F' || suf === '.SG' || suf === '.BE') return 'Allemagne'; if (suf === '.MI') return 'Italie'; if (suf === '.MC') return 'Espagne'; if (suf === '.BR') return 'Belgique'; if (suf === '.SW') return 'Suisse'; if (suf === '.TO' || suf === '.V') return 'Canada'; if (suf === '.AX' || suf === '.NZ' || suf === '.HK' || suf === '.T' || suf === '.SS' || suf === '.SZ' || suf === '.KS' || suf === '.KQ') return 'Asie-Pacifique'; return 'Monde'; } // Secteur et zone d'une action. Double tentative : API de recherche Yahoo // (la meme que "Chercher un ticker"), puis page profil en secours. // Pour les codes ABC (mnemoniques Euronext) : traduction en ticker Yahoo // (.PA) — on n'essaie JAMAIS le mnemonique brut, qui designerait un autre // actif sur Yahoo (ex : SAN tout court = Santander). function fetchProfilYahoo(code, source) { try { var essais = []; var cu = String(code).toUpperCase(); if (source === 'ABC') { if (CODES_YAHOO[cu]) essais.push(CODES_YAHOO[cu]); essais.push(cu + '.PA'); } else { essais.push(code); } var secteur = '', zone = ''; for (var i = 0; i < essais.length && !secteur; i++) { var t1 = profilViaRecherche(essais[i]); if (t1) { secteur = t1.secteur; zone = t1.zone; } } if (!secteur) { for (var k = 0; k < essais.length && !secteur; k++) { var t2 = profilViaPage(essais[k]); if (t2) { secteur = t2.secteur; zone = t2.zone; } } } if (!secteur) return null; return { secteur: secteur, zone: zone }; } catch (e) { return null; } } function profilViaRecherche(ticker) { try { var r = UrlFetchApp.fetch( 'https://query1.finance.yahoo.com/v1/finance/search?q=' + encodeURIComponent(ticker) + '"esCount=5&newsCount=0', { muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' } }); if (r.getResponseCode() !== 200) return null; var quotes = (JSON.parse(r.getContentText()).quotes) || []; for (var j = 0; j < quotes.length; j++) { var q = quotes[j]; if (String(q.symbol || '').toUpperCase() === String(ticker).toUpperCase() && q.sector) { return { secteur: TRAD_SECTEURS[q.sector] || q.sector, zone: zoneDepuisSuffixe(q.symbol) }; } } return null; } catch (e) { return null; } } function profilViaPage(ticker) { try { var r = UrlFetchApp.fetch('https://finance.yahoo.com/quote/' + encodeURIComponent(ticker) + '/profile', { muteHttpExceptions: true, followRedirects: true, headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' } }); if (r.getResponseCode() !== 200) return null; var txt = r.getContentText(); var ms = txt.match(/"sector"\s*:\s*"([^"]+)"/); if (!ms) return null; var secteur = TRAD_SECTEURS[ms[1]] || ms[1]; var mc = txt.match(/"country"\s*:\s*"([^"]+)"/); var pays = mc ? mc[1] : ''; if (pays === 'United States') pays = 'USA'; else if (pays === 'United Kingdom') pays = 'Royaume-Uni'; else if (!pays) pays = zoneDepuisSuffixe(ticker); return { secteur: secteur, zone: pays }; } catch (e) { return null; } } function completerEtiquettes() { var cfgSh = getSS().getSheetByName(SHEET_CFG); if (!cfgSh) { SpreadsheetApp.getUi().alert('Lance d\'abord "1. Initialiser".'); return; } var last = cfgSh.getLastRow(); if (last < 2) return; var vals = cfgSh.getRange(2, 1, last - 1, 9).getValues(); var rapport = []; for (var i = 0; i < vals.length; i++) { var nom = String(vals[i][1] || '').trim(); if (!nom || nom.toUpperCase().indexOf('EXEMPLE') !== -1) continue; var sec = '', zone = ''; var code = String(vals[i][2] || '').trim(); var src = String(vals[i][3] || 'YAHOO').trim().toUpperCase() || 'YAHOO'; var cu = code.toUpperCase(); if (cu.indexOf('VIDE') !== -1 || cu.indexOf('VERIFIER') !== -1 || cu.indexOf('AJUSTER') !== -1) code = ''; if (!code) { // Actif sans cotation : reconnaissance par nom (cash, fonds euros, livret A) var nu = nom.toUpperCase(); if (nu.indexOf('LIVRET') !== -1 || nu.indexOf('ESPECES') !== -1 || nu.indexOf('ESPECE') !== -1 || nu.indexOf('ESPÈCES') !== -1 || nu.indexOf('CASH') !== -1) { sec = 'Cash'; // le cash sera exclu des camemberts Secteurs / Zones } else if (nu.indexOf('FONDS') !== -1 || nu.indexOf('EURO') !== -1) { sec = 'Obligations'; zone = 'France'; } } else { var p = profilETF(nom, code); if (p) { sec = p.sec; zone = p.zone; } else { var r = fetchProfilYahoo(code, src); if (r) { sec = r.secteur; zone = r.zone; } } } var r8 = 2 + i; var rempli = false; if (sec && !String(vals[i][7] || '').trim()) { cfgSh.getRange(r8, 8).setValue(sec); rempli = true; } if (zone && !String(vals[i][8] || '').trim()) { cfgSh.getRange(r8, 9).setValue(zone); rempli = true; } if (rempli) rapport.push('OK ' + nom + ' -> ' + sec + ' | ' + zone); else if (!String(vals[i][7] || '').trim() || !String(vals[i][8] || '').trim()) { rapport.push('? ' + nom + ' : non reconnu (code : ' + (code || 'VIDE') + ', source : ' + src + ') -> remplis les colonnes H et I à la main'); } } try { SpreadsheetApp.getUi().alert( 'SECTEURS ET ZONES (v' + VERSION_SCRIPT + ')\n\n' + (rapport.length ? rapport.join('\n') : 'Rien à compléter : tout est déjà renseigné.') + '\n\nRelance ensuite "2. Reconstruire le tableau".\n' + 'Tu peux ajuster à la main : un libellé simple ou une répartition pondérée.' ); } catch (e) { Logger.log(rapport.join('\n')); } } // ---------------- RECHERCHE DE TICKER ------------------------------------- function chercherTicker() { var ui = SpreadsheetApp.getUi(); var res = ui.prompt( 'Chercher un ticker', 'Tape un nom ou un code ISIN (ex : "Vinci", "IE00BYX2JD69") :', ui.ButtonSet.OK_CANCEL ); if (res.getSelectedButton() !== ui.Button.OK) return; try { var resp = UrlFetchApp.fetch( 'https://query1.finance.yahoo.com/v1/finance/search?q=' + encodeURIComponent(res.getResponseText()) + '"esCount=8&newsCount=0', { muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' } }); var data = JSON.parse(resp.getContentText()); var lines = (data.quotes || []).map(function (q) { return q.symbol + ' — ' + (q.shortname || q.longname || '') + ' (' + (q.exchange || '') + ')'; }); ui.alert( lines.length ? 'Copie le ticker voulu dans la colonne C de Configuration :\n\n' + lines.join('\n') : 'Aucun résultat. Essaie un autre nom.' ); } catch (e) { ui.alert('Recherche impossible : ' + e.message); } } function afficherModeDEmploi() { var help = getSS().getSheetByName(SHEET_HELP); if (help) help.activate(); else SpreadsheetApp.getUi().alert("Lance d'abord \"1. Initialiser\"."); } // ---------------- DIAGNOSTIC DES GRAPHIQUES ------------------------------- function diagnosticGraphiques() { var ss = getSS(); var out = []; [SHEET_MAIN, SHEET_EV].forEach(function (nom) { var sh = ss.getSheetByName(nom); if (!sh) { out.push(nom + ' : feuille introuvable'); return; } var cs = sh.getCharts(); out.push(nom + ' : ' + cs.length + ' graphique(s)'); for (var i = 0; i < cs.length; i++) { out.push(' - ' + cs[i].getType() + ' (titre : ' + (cs[i].getOptions()['title'] || 'sans titre') + ')'); } }); SpreadsheetApp.getUi().alert( 'ETAT DES GRAPHIQUES\n\n' + out.join('\n') + '\n\nS\'il y a 0 graphique ici alors que la reconstruction disait OK, copie ce message tel quel.' ); }
